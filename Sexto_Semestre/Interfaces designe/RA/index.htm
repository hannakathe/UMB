<!-- Esto es todo lo que hace falta para tener RA en una página -->
<script type="module" src="https://cdn.jsdelivr.net/npm/@google/model-viewer@4.4.0/dist/model-viewer.min.js"></script>

<model-viewer
  src="Astronaut.glb"  <!-- modelo 3D en glTF/GLB -->
  alt="Modelo 3D de un astronauta"  <!-- accesibilidad -->
  ar  <!-- activa el botón de RA -->
  ar-modes="webxr scene-viewer quick-look" <!-- orden de preferencia -->
  ar-placement="floor"  <!-- piso o pared -->
  camera-controls auto-rotate shadow-intensity="1">
</model-viewer>
Modelos de ejemplo: shared-assets de modelviewer.dev · Para probar la RA hay que abrir la página por HTTPS (por ejemplo, GitHub Pages) desde un celular compatible.
